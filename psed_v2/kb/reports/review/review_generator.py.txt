def wr(name, rows, fields=None):
    rows = list(rows)

        w = csv.DictWriter(fh, fieldnames=fields, extrasaction="ignore")
        w.writeheader()
        w.writerows(rows)
    return len(rows)

def cast(r):

    rec["quote"] = " ".join(r["statement"].split()[:15]) or "-"

def ckey(e):
    """A jsonschema error -> the constraint it hits, in the scheme of the constraint table."""
    sp = [str(x) for x in e.schema_path]
    if e.validator == "required":
        if sp[0] == "allOf":
            return f"allOf[{sp[1]}].{sp[2]}"
        missing = re.findall(r"'([^']+)'", e.message)[0]
        return f"required:{'.'.join([x for x in sp[:-1] if x != 'properties'] + [missing])}"
    if e.validator == "dependentRequired":
        a = re.findall(r"'([^']+)'", e.message)
        return f"dependentRequired:{a[1]}->{a[0]}"
    if sp and sp[0] == "allOf":
        return f"allOf[{sp[1]}].{sp[2]}"
    path = ".".join(x for x in sp[:-1] if x not in ("properties", "items", "additionalProperties", "$ref") and not x.isdigit())
    v = e.validator if e.validator != "False schema" and e.schema is not False else "forbidden"
    return f"{v}:{path}"


def constraints(schema, name):
    out = []
    for f in schema.get("required", []):
        out.append(dict(schema=name, constraint=f"required:{f}", kind="required", detail=f"{f} always"))
    for k, vs in schema.get("dependentRequired", {}).items():
        for v in vs:
            out.append(dict(schema=name, constraint=f"dependentRequired:{k}->{v}", kind="dependentRequired", detail=f"{v} required with {k}"))
    for i, a in enumerate(schema.get("allOf", [])):
        for br in ("then", "else"):
            if br in a:
                out.append(dict(schema=name, constraint=f"allOf[{i}].{br}", kind="allOf if-then", detail=f"if {json.dumps(a['if'])} {br} {json.dumps(a[br])}"[:300]))

    def walk(s, path):
        if not isinstance(s, dict):
            return
        if s.get("additionalProperties") is False:
            out.append(dict(schema=name, constraint=f"additionalProperties:{path}", kind="additionalProperties", detail=f"no keys outside {sorted(s.get('properties', {}))}"[:300]))
        for kw in ("pattern", "minLength", "minItems", "uniqueItems", "minProperties", "enum", "type"):
            if kw in s and path:
                out.append(dict(schema=name, constraint=f"{kw}:{path}", kind=kw, detail=(f"{len(s[kw])} values" if kw == "enum" else str(s[kw]))[:200]))
        for k, sub in (s.get("properties") or {}).items():
            walk(sub, f"{path}.{k}" if path else k)
        if isinstance(s.get("items"), dict):
            walk(s["items"], path)
        if isinstance(s.get("additionalProperties"), dict):
            walk(s["additionalProperties"], path)
    walk(schema, "")
    return out
